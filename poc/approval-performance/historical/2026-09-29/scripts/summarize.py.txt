import collections, gzip, hashlib, json, math, re
from pathlib import Path
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parents[2]
HERE = Path(__file__).resolve().parent

def digest(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for block in iter(lambda: f.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()

def read_json(path):
    if path.exists():
        return json.loads(path.read_text())
    with gzip.open(str(path) + '.gz', 'rt') as f:
        return json.load(f)

def metrics(path):
    values = {}
    if not path.exists():
        return values
    for line in path.read_text().splitlines():
        if not line or line.startswith('#'):
            continue
        match = re.fullmatch(r'([^ {]+)(?:\{([^}]+)\})?\s+([^ ]+)', line)
        if not match:
            continue
        name, label_string, value = match.groups()
        labels = dict(re.findall(r'(\w+)="([^"]*)"', label_string or ''))
        labels.pop('target', None)
        key = (name, tuple(sorted(labels.items())))
        values[key] = values.get(key, 0) + float(value)
    return values

def metric_rows(values, prefix):
    return [{'name': n, 'labels': dict(labels), 'value': v}
            for (n, labels), v in sorted(values.items()) if n.startswith(prefix)]

cases = []
for name in ['jdk', 'bouncy-castle', 'approvals-off']:
    directory = HERE / 'performance' / name
    report = read_json(directory / 'transactions-report.json')
    assert not report.get('error'), report.get('error')
    before, after = [metrics(directory / ('metrics-' + phase + '.txt')) for phase in ['before', 'after']]
    delta = {k: v - before.get(k, 0) for k, v in after.items()}
    old_node, node = [metrics(directory / ('besu-metrics-' + phase + '.txt')) for phase in ['before', 'after']]
    node_delta = {k: v - old_node.get(k, 0) for k, v in node.items()}
    def total(family):
        return sum(v for (n, _), v in delta.items() if n == 'privacyproxy_approval_' + family)
    count = total('delivery_seconds_count')
    buckets = sorted((float(dict(labels)['le']), v) for (n, labels), v in delta.items()
                     if n == 'privacyproxy_approval_delivery_seconds_bucket')
    latency = None if not count else {'mean_ms': total('delivery_seconds_sum') * 1000 / count,
        'histogram_count': count,
        'cumulative_buckets_seconds': {str(b): n for b, n in buckets},
        'percentile_upper_bounds_ms': {p: next(b for b, n in buckets if n >= math.ceil(q * count)) * 1000
                                      for p, q in [('p50', .5), ('p95', .95), ('p99', .99)]}}
    receipts = read_json(directory / 'receipts.json')
    logs = read_json(directory / 'transactions.json')
    missing = {row['hash'] for row in receipts if row['receipt'] is None}
    statuses = {r['txHash'].lower(): r.get('status', 'unknown') for r in logs if r.get('txHash')}
    errors = collections.Counter(r.get('errorMessage') or r.get('error') for r in logs
                                 if r.get('errorMessage') or r.get('error'))
    cases.append({'case': name, 'config': read_json(directory / 'verifier-config.json'),
        'receipts': report['receipts'], 'generator': {k: report['status'].get(k)
            for k in ['txSent', 'txConfirmed', 'txFailed', 'status', 'elapsedMs']},
        'missing_by_generator_status': dict(collections.Counter(statuses.get(tx, 'unknown') for tx in missing)),
        'generator_error_counts': dict(errors), 'approval_delivery': latency,
        'mean_approvals_per_batch': total('confirmed_total') / count if count else None,
        'approval_counter_deltas': metric_rows({k:v for k,v in delta.items() if k[0].endswith('_total')}, 'privacyproxy_approval_'),
        'producer_counter_deltas': metric_rows({k:v for k,v in node_delta.items() if k[0].endswith('_total')}, 'ops_approval_'),
        'producer_final_gauges': metric_rows({k:v for k,v in node.items() if not k[0].endswith('_total')}, 'ops_approval_'),
        'loadgen_sha256': report['loadgen_sha256'],
        'source_files_sha256': {str(p.relative_to(directory)): digest(p) for p in directory.iterdir()
                                if p.is_file() and p.suffix in ['.json', '.txt', '.gz']},
        'memory_sampling_error': report.get('memory_sampling_error')})
